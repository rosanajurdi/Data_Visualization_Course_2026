# Liste des mots vides (stopwords) en anglais à filtrer
mots_vides = set(stopwords.words('english'))

# Fonction pour adapter les tags au format de WordNet
def obtenir_pos_wordnet(tag):
    if tag.startswith('J'):
        return 'a' # Adjectifs
    if tag.startswith('V'):
        return 'v' # Verbes
    if tag.startswith('N'):
        return 'n' # Noms
    if tag.startswith('R'):
        return 'r' # Adverbes
    return 'n'     # Noms par défaut

tous_lemmes = []
tous_tags_pos = []

# Traitement, nettoyage et lemmatisation pour chaque texte
for texte in textes_avis:
    # On enlève tout ce qui n'est pas une lettre et on passe en minuscules
    texte_propre = re.sub(r'[^a-zA-Z\s]', ' ', texte.lower())
    texte_propre = re.sub(r'\s+', ' ', texte_propre).strip()
    
    if texte_propre == '':
        continue

    # On parcours chaque mot tagué pour le vérifier et le lemmatiser
    for mot, tag in TextBlob(texte_propre).tags:
        if mot in mots_vides or len(mot) <= 1:
            continue
            
        # Extraction du lemme
        lemme = Word(mot).lemmatize(obtenir_pos_wordnet(tag))
        tous_lemmes.append(lemme)
        tous_tags_pos.append(tag)

print('Nombre total de tokens lemmatisés:', len(tous_lemmes))
