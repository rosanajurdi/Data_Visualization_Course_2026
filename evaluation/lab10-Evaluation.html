# Question 1: Sentiment Analysis + Bar Chart

def classify_sentiment(text):
    polarity = TextBlob(str(text)).sentiment.polarity
    if polarity > 0.1:
        return 'positive'
    elif polarity < -0.1:
        return 'negative'
    else:
        return 'neutral'

df['polarity']  = df['reviewText'].apply(lambda t: TextBlob(str(t)).sentiment.polarity)
df['sentiment'] = df['reviewText'].apply(classify_sentiment)

sentiment_counts = df['sentiment'].value_counts().reset_index()
sentiment_counts.columns = ['sentiment', 'count']

print('Sentiment distribution:')
for _, row in sentiment_counts.iterrows():
    pct = row['count'] / len(df) * 100
    print(f'  {row["sentiment"]:10}: {row["count"]:3} reviews ({pct:.1f}%)')

# Bar Chart
fig, ax = plt.subplots(figsize=(8, 5))
colors = {'positive': '#4CAF50', 'negative': '#F44336', 'neutral': '#FFC107'}
bar_colors = [colors.get(s, 'steelblue') for s in sentiment_counts['sentiment']]

bars = ax.bar(
    sentiment_counts['sentiment'],
    sentiment_counts['count'],
    color=bar_colors, edgecolor='white', linewidth=1.5, width=0.5
)

# Value labels on bars
for bar, (_, row) in zip(bars, sentiment_counts.iterrows()):
    pct = row['count'] / len(df) * 100
    ax.text(
        bar.get_x() + bar.get_width() / 2,
        bar.get_height() + 0.3,
        f"{row['count']} ({pct:.0f}%)",
        ha='center', va='bottom', fontweight='bold', fontsize=11
    )

ax.set_title('Sentiment Distribution of Amazon Reviews', fontsize=15, fontweight='bold', pad=15)
ax.set_xlabel('Sentiment Category', fontsize=12)
ax.set_ylabel('Number of Reviews', fontsize=12)
ax.set_ylim(0, sentiment_counts['count'].max() * 1.2)
plt.tight_layout()
plt.show()

# Question 2: Text Preprocessing

stop_words = set(nltk_sw.words('english')) | set(STOPWORDS)

def get_wordnet_pos(treebank_tag):
    """Map Treebank POS tag to WordNet POS tag."""
    if treebank_tag.startswith('J'): return 'a'   # Adjective
    elif treebank_tag.startswith('V'): return 'v' # Verb
    elif treebank_tag.startswith('N'): return 'n' # Noun
    elif treebank_tag.startswith('R'): return 'r' # Adverb
    else: return 'n'

def preprocess_and_lemmatize(text):
    """Full preprocessing pipeline: lowercase -> punctuation/number removal
    -> stopword filtering -> POS-aware lemmatization."""
    # 1. Lowercase
    text = str(text).lower()
    # 2. Remove punctuation and numbers
    text = re.sub(r'[^a-z\s]', '', text)
    # 3. Tokenize and POS tag
    blob = TextBlob(text)
    lemmatized = []
    for word, tag in blob.tags:
        word_clean = re.sub(r'[^a-z]', '', word)
        # 4. Filter stopwords and very short words
        if word_clean and len(word_clean) > 2 and word_clean not in stop_words:
            # 5. Lemmatize with correct POS
            wn_pos = get_wordnet_pos(tag)
            lemma = Word(word_clean).lemmatize(wn_pos)
            lemmatized.append(lemma)
    return lemmatized

# Apply to all reviews
df['tokens'] = df['reviewText'].apply(preprocess_and_lemmatize)

# Question 5: POS Frequency - Nouns, Verbs, Adjectives

nouns, verbs, adjectives = [], [], []

for review in df['reviewText']:
    text_clean = re.sub(r'[^a-z\s]', '', str(review).lower())
    blob = TextBlob(text_clean)
    for word, tag in blob.tags:
        word_clean = re.sub(r'[^a-z]', '', word)
        if not word_clean or len(word_clean) < 3 or word_clean in stop_words:
            continue
        lemma = Word(word_clean).lemmatize(get_wordnet_pos(tag))
        if tag.startswith('NN'):   nouns.append(lemma)
        elif tag.startswith('VB'): verbs.append(lemma)
        elif tag.startswith('JJ'): adjectives.append(lemma)

top_nouns = Counter(nouns).most_common(12)
top_verbs = Counter(verbs).most_common(12)
top_adjs  = Counter(adjectives).most_common(12)

print('POS statistics:')
print(f'  Nouns      : {len(nouns):4} tokens, {len(set(nouns)):4} unique')
print(f'  Verbs      : {len(verbs):4} tokens, {len(set(verbs)):4} unique')
print(f'  Adjectives : {len(adjectives):4} tokens, {len(set(adjectives)):4} unique')

# Visualization
fig, axes = plt.subplots(1, 3, figsize=(18, 6))
fig.suptitle('POS Frequency Analysis - Amazon Reviews\n(Nouns, Verbs, Adjectives)',
             fontsize=15, fontweight='bold')

pos_config = [
    (axes[0], top_nouns,  'Top 12 Nouns',      'darkorange',  f'{len(nouns)} total'),
    (axes[1], top_verbs,  'Top 12 Verbs',      'royalblue',   f'{len(verbs)} total'),
    (axes[2], top_adjs,   'Top 12 Adjectives', 'seagreen',    f'{len(adjectives)} total'),

for ax, top_words, title, color, subtitle in pos_config:

    ax.set_title(f'{title}\n({subtitle})', fontweight='bold')

    for i, v in enumerate(list(cs)[::-1]):
        ax.text(v + 0.1, i, str(v), va='center', fontsize=9, fontweight='bold')
