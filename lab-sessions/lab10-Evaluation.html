# 1. Analyse des Sentiments
from textblob import TextBlob
import matplotlib.pyplot as plt
import seaborn as sns

def get_sentiment(text):
    # Sécurité au cas où il y a des cellules vides
    if not isinstance(text, str):
        return 'neutral'
    
    # TextBlob analyse le sentiment (Polarity va de -1 à 1)
    polarity = TextBlob(text).sentiment.polarity
    if polarity > 0:
        return 'positive'
    elif polarity < 0:
        return 'negative'
    else:
        return 'neutral'

print("Analyse des sentiments en cours...")
df['Sentiment'] = df['reviewText'].apply(get_sentiment)

# Création du graphique à barres
plt.figure(figsize=(8, 5))
sns.countplot(data=df, x='Sentiment', hue='Sentiment', palette='viridis', order=['positive', 'negative', 'neutral'], legend=False)
plt.title('Distribution des Sentiments des Clients')
plt.xlabel('Sentiment')
plt.ylabel('Nombre d\'avis')
plt.show()

# 2. Pipeline de Prétraitement NLP
import re
import nltk
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer
from nltk.corpus import wordnet

lemmatizer = WordNetLemmatizer()
stop_words = set(stopwords.words('english'))

# Convertir les catégories grammaticales pour le Lemmatizer
def get_wordnet_pos(treebank_tag):
    if treebank_tag.startswith('J'):
        return wordnet.ADJ
    elif treebank_tag.startswith('V'):
        return wordnet.VERB
    elif treebank_tag.startswith('N'):
        return wordnet.NOUN
    elif treebank_tag.startswith('R'):
        return wordnet.ADV
    else:
        return wordnet.NOUN

def preprocess_text(text):
    if not isinstance(text, str):
        return ""
    # a. Minuscules
    text = text.lower()
    # b. Suppression de la ponctuation et des chiffres
    text = re.sub(r'[^a-z\s]', '', text)
    # Tokenisation
    words = text.split()
    # c. POS Tagging
    pos_tags = nltk.pos_tag(words)
    
    clean_words = []
    # d. Stopwords et Lemmatisation
    for word, tag in pos_tags:
        if word not in stop_words:
            wn_pos = get_wordnet_pos(tag)
            lemma = lemmatizer.lemmatize(word, pos=wn_pos)
            clean_words.append(lemma)
            
    return " ".join(clean_words)

print("Nettoyage intensif en cours")
df['Clean_Text'] = df['reviewText'].apply(preprocess_text)
print("Nettoyage terminé !")

# 5. Fréquence des catégories grammaticales (POS)
print("Comptage des Noms, Verbes et Adjectifs en cours...")

def count_pos(text):
    if not isinstance(text, str):
        return 0, 0, 0
    # Tokenisation rapide
    words = text.split()
    tags = nltk.pos_tag(words)
    
    # Comptage via les tags NLTK (N=Nom, V=Verbe, J=Adjectif)
    n = sum(1 for word, tag in tags if tag.startswith('N'))
    v = sum(1 for word, tag in tags if tag.startswith('V'))
    a = sum(1 for word, tag in tags if tag.startswith('J'))
    return n, v, a

# On applique la fonction et on additionne les totaux
pos_counts = df['Clean_Text'].apply(count_pos)

noun_total = sum(x[0] for x in pos_counts)
verb_total = sum(x[1] for x in pos_counts)
adj_total = sum(x[2] for x in pos_counts)

# Création du DataFrame pour le graphique
pos_df = pd.DataFrame({
    'Type': ['Noms (Nouns)', 'Verbes (Verbs)', 'Adjectifs (Adjectives)'],
    'Frequence': [noun_total, verb_total, adj_total]
})

plt.figure(figsize=(8, 5))
sns.barplot(data=pos_df, x='Type', y='Frequence', hue='Type', palette='autumn', legend=False)
plt.title('Distribution des Noms, Verbes et Adjectifs')
plt.ylabel('Nombre total d\'apparitions')
plt.show()
